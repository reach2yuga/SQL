- Order by clause required
- Partition by clause is optional
- ROW_NUMBER() over (Order by col1,col2)

Rank():

- Return rank starting from 1 when partion changed its value reset to 1
- oder by clause required
- Partition by clause is optional
- When there is tie it skips rank


Dense_Rank():

- Return rank starting from 1 when partion changed its value reset to 1
- oder by clause required
- Partition by clause is optional
- When there is tie it will not skip the rank

Usage Case: Both rank(),denserank() function used to find nth highest salary 
            when there is tie

-- Row number, rank, and dense rank by salary
SELECT 
    emp_name,
    department,
    salary,
    ROW_NUMBER() OVER(ORDER BY salary DESC) AS row_num,
    RANK() OVER(ORDER BY salary DESC) AS rank,
    DENSE_RANK() OVER(ORDER BY salary DESC) AS dense_rank
FROM employees;

 SELECT 
    emp_name, 
    salary,
    ROW_NUMBER() OVER(ORDER BY salary DESC) AS rank
FROM employees;


-- What if two employees have the same salary?
SELECT 
    emp_name, 
    salary,
    RANK() OVER(ORDER BY salary DESC) AS rank
FROM employees;

SELECT 
    emp_name, 
    salary,
    DENSE_RANK() OVER(ORDER BY salary DESC) AS rank
FROM employees;

SELECT 
    emp_name, 
    salary,
    NTILE(4) OVER(ORDER BY salary) AS quartile
FROM employees;


--NTILE(4) 
--NTILE(4) is a window function that divides rows into approximately
--equal groups (quartiles in this case). When used with PARTITION BY department, 
--it creates these groups separately for each department.

--Handling Uneven Groups
--When employee counts aren't divisible by 4:
--The first groups get extra members
--Example: 5 employees → Quartiles: 2, 2, 1, 1
--Example: 6 employees → Quartiles: 2, 

SELECT 
    emp_name,
    department,
    salary,
    NTILE(3) OVER(
        PARTITION BY department 
        ORDER BY salary DESC
    ) AS salary_quartile
FROM employees;
