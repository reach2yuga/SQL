-- Step 5: INNER JOIN
SELECT A.value AS A_value, B.value AS B_value
FROM TableA A
INNER JOIN TableB B
    ON A.value = B.value; //6

-- Step 6: LEFT JOIN
SELECT A.value AS A_value, B.value AS B_value
FROM TableA A
LEFT JOIN TableB B
    ON A.value = B.value; //8


-- RIGHT JOIN
SELECT A.value AS A_value, B.value AS B_value
FROM TableA A
RIGHT JOIN TableB B
    ON A.value = B.value; //9


-- FULL OUTER JOIN
SELECT A.value AS A_value, B.value AS B_value
FROM TableA A
FULL OUTER JOIN TableB B
    ON A.value = B.value; //11
